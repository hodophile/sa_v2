# requirements.app.txt
# sa_v2 / VideoPrism — FastAPI app, NVIDIA GPU (Linux x86_64, CUDA 12, Py 3.10–3.13)
#
# The model runs on JAX + GPU. `tensorflow-cpu` is here ONLY because
# videoprism/{tokenizers,utils}.py use `from tensorflow.io import gfile` and
# `tf.py_function` / `tf.RaggedTensor` for the text tokenizer. CPU TF is the
# right choice even on a GPU box: the full `tensorflow` GPU wheel ships ~3 GB
# of NVIDIA CUDA libs that can ABI-conflict with JAX's CUDA stack, and it is
# not involved in model compute. The crash you saw
# (`_pywrap_tensorflow_lite_metrics_wrapper.so: undefined symbol`) was caused
# by `protobuf` 7.x being pulled in; TF 2.21.0's compiled wheels were built
# against the protobuf 6.31.x line, so protobuf is pinned to <7.
#
# Platform: Linux x86_64 + CUDA 12.x + NVIDIA driver >= 525.
#   - `jax-cuda12-plugin` / `-pjrt` ship only manylinux x86_64 & aarch64
#     wheels → this file is Linux-only. On Windows/macOS use `jax` (CPU) or
#     build JAX CUDA from source.
#   - Python 3.10–3.13 only (tensorflow-cpu==2.21.0 has no cp39/cp314 wheels).

# ── JAX / GPU stack (pinned to the line videoprism checkpoints were built on) ──
jax==0.11.1
jaxlib==0.11.1
jax-cuda12-plugin==0.11.1
jax-cuda12-pjrt==0.11.1

# ── Flax (model definitions + checkpoint loading) — pulls in optax & orbax ──
flax==0.12.9
optax==0.2.8               # transitive runtime dep of flax 0.12.9
orbax-checkpoint==0.12.4   # transitive runtime dep of flax 0.12.9

# ── Array / numerics (jaxlib 0.11.1 requires numpy>=2.0, scipy>=1.14, ml_dtypes>=0.5) ──
numpy==2.1.3
scipy>=1.14
ml_dtypes==0.6.0

# ── Video / image I/O ──
mediapy==1.2.7
opencv-python==5.0.0.93     # `cv2` in app.py

# ── TF (CPU only — see header) ──
tensorflow-cpu==2.21.0
protobuf>=6.31.1,<7        # TF 2.21 was built against protobuf 6.31.x; <7 avoids the
                           # _pywrap_tensorflow_lite_metrics_wrapper.so undefined-symbol crash

# ── Tokenizer / model assets ──
sentencepiece==0.2.2
huggingface-hub==1.29.0
einops==0.8.2

# ── FastAPI service ──
